```java
package com.edgechains.lib.utils;

import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.comprehend.ComprehendClient;
import software.amazon.awssdk.services.comprehend.model.DetectPiiEntitiesRequest;
import software.amazon.awssdk.services.comprehend.model.DetectPiiEntitiesResponse;
import software.amazon.awssdk.services.comprehend.model.LanguageCode;
import software.amazon.awssdk.services.comprehend.model.PiiEntity;
import software.amazon.awssdk.services.comprehend.model.PiiEntityType;

import java.util.*;

/**
 * Production-ready AWS Comprehend Utility for EdgeChains to detect and redact sensitive PII.
 */
public class AwsComprehendRedactionUtility implements AutoCloseable {

    private final ComprehendClient comprehendClient;
    private final boolean ownsClient;

    public enum MaskStyle {
        ENTITY_TYPE,    // [NAME], [EMAIL], [SSN]
        REDACTED_TAG,   // [REDACTED]
        ASTERISKS       // ***
    }

    public static class RedactionOptions {
        private final LanguageCode languageCode;
        private final double minConfidence;
        private final MaskStyle maskStyle;
        private final Set<PiiEntityType> filterTypes;

        public RedactionOptions(LanguageCode languageCode, double minConfidence, MaskStyle maskStyle, Set<PiiEntityType> filterTypes) {
            this.languageCode = languageCode != null ? languageCode : LanguageCode.EN;
            this.minConfidence = minConfidence;
            this.maskStyle = maskStyle != null ? maskStyle : MaskStyle.ENTITY_TYPE;
            this.filterTypes = filterTypes != null ? filterTypes : Collections.emptySet();
        }

        public static RedactionOptions defaultOptions() {
            return new RedactionOptions(LanguageCode.EN, 0.65, MaskStyle.ENTITY_TYPE, Collections.emptySet());
        }
    }

    public AwsComprehendRedactionUtility() {
        this(ComprehendClient.builder()
                .credentialsProvider(DefaultCredentialsProvider.create())
                .region(Region.US_EAST_1)
                .build(), true);
    }

    public AwsComprehendRedactionUtility(ComprehendClient client) {
        this(client, false);
    }

    private AwsComprehendRedactionUtility(ComprehendClient client, boolean ownsClient) {
        this.comprehendClient = Objects.requireNonNull(client, "ComprehendClient must not be null");
        this.ownsClient = ownsClient;
    }

    /**
     * Redacts PII entities in the source text while preserving text outside matching offsets.
     */
    public String redact(String text, RedactionOptions options) {
        if (text == null || text.trim().isEmpty()) {
            return text;
        }

        DetectPiiEntitiesRequest request = DetectPiiEntitiesRequest.builder()
                .text(text)
                .languageCode(options.languageCode)
                .build();

        DetectPiiEntitiesResponse response = comprehendClient.detectPiiEntities(request);
        List<PiiEntity> entities = new ArrayList<>(response.entities());

        if (entities.isEmpty()) {
            return text;
        }

        // 必须按 beginOffset 降序倒排，保证前置字符串切片索引不被后续替换长度影响
        entities.sort(Comparator.comparingInt(PiiEntity::beginOffset).reversed());

        StringBuilder sb = new StringBuilder(text);
        int lastStart = Integer.MAX_VALUE;

        for (PiiEntity entity : entities) {
            if (entity.score() != null && entity.score() < options.minConfidence) {
                continue;
            }
            if (!options.filterTypes.isEmpty() && !options.filterTypes.contains(entity.type())) {
                continue;
            }

            int start = entity.beginOffset();
            int end = entity.endOffset();

            // 防重叠边界截断
            if (end > lastStart) {
                end = lastStart;
            }
            if (start >= end || start < 0 || end > sb.length()) {
                continue;
            }

            String replacement;
            switch (options.maskStyle) {
                case REDACTED_TAG:
                    replacement = "[REDACTED]";
                    break;
                case ASTERISKS:
                    replacement = "*".repeat(Math.max(3, end - start));
                    break;
                case ENTITY_TYPE:
                default:
                    replacement = "[" + entity.typeAsString() + "]";
                    break;
            }

            sb.replace(start, end, replacement);
            lastStart = start;
        }

        return sb.toString();
    }

    @Override
    public void close() {
        if (ownsClient) {
            comprehendClient.close();
        }
    }
}
```

### What alternative solutions did you explore? (Optional)

1. **State Persistence via Git Commits (`bounties_seen.json`)**:
   - *Exploration*: Committing a tracked JSON file to Git on every run.
   - *Trade-off*: Would trigger recursive workflow executions or require `[skip ci]` commit discipline and elevated repo write permissions. Leveraging GitHub Issues with deterministic titles and label guards provides a cleaner, serverless state store.
2. **Hard-Filtering `🔴` Out of Digest Bodies Entirely**:
   - *Exploration*: Omitting `🔴` items from both the digest list and markdown markers.
   - *Trade-off*: Dropping them before marking them as seen would cause the search loop to re-fetch and re-assess them on every run. Keeping them in the hidden markers (`<!-- algora:gh-... -->`) while excluding them from `actionable` guarantees zero-waste deduplication.
